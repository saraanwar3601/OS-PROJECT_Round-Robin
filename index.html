Student Assignment Scaffold: Modifying FCFS to Round Robin (RR)

Where do students write code?

The scheduler logic is located in the JavaScript function executeScheduler(initialProcs, csOverhead) inside this single HTML file. In the baseline, it iterates non-preemptively. To implement Round Robin, students must add time quantum slicing and a dynamic FIFO ready queue.

// =========================================================================
// STUDENT ASSIGNMENT: ROUND ROBIN (RR) CONVERSION BLUEPRINT
// =========================================================================

function executeRoundRobin(initialProcs, quantum = 10.0, csOverhead = 1.0) {
// 1. Initialize FIFO readyQueue and tracking variables
let currentTime = 0.0;
const readyQueue = [];
const completed = [];
let procIdx = 0;

// 2. Enqueue arrived processes helper
const enqueueArrived = (timeLimit) => {
while (procIdx < procs.length && procs[procIdx].arrival_time <= timeLimit) {
readyQueue.push(procs[procIdx]);
procIdx++;
}
};

// 3. Main Round Robin execution loop
while (completed.length < procs.length) {
// If ready queue is empty, advance clock across idle gap
if (readyQueue.length === 0 && procIdx < procs.length) {
currentTime = procs[procIdx].arrival_time;
enqueueArrived(currentTime);
}

// Pop front process from ready queue:
const curr = readyQueue.shift();
if (curr.start_time === null) curr.start_time = currentTime; // Lock RT

// Compute slice duration: min(remaining_time, quantum)
const slice = Math.min(curr.remaining_time, quantum);
currentTime += slice;
curr.remaining_time -= slice;

// CRITICAL: Enqueue arrivals that occurred DURING the active burst slice
enqueueArrived(currentTime);

// Incur 1.0 ms context switch overhead:
if (curr.remaining_time <= 0) {
// Process finished
curr.finish_time = currentTime;
completed.push(curr);
currentTime += csOverhead; // 1.0 ms transition overhead
enqueueArrived(currentTime);
} else {
// Quantum expired (Preemption)
currentTime += csOverhead; // 1.0 ms preemption overhead
enqueueArrived(currentTime);
readyQueue.push(curr); // Re-queue at the tail
}
}
return { completed, makespan: currentTime };
}
